cmake_minimum_required(VERSION 3.24)
project(libtcod_package_test LANGUAGES C)

option(TEST_EXISTING_ALIAS "Load the package with an existing namespaced alias." OFF)

if(TEST_EXISTING_ALIAS)
    add_library(existing INTERFACE)
    add_library(libtcod::libtcod ALIAS existing)
    target_include_directories(existing INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}")
    get_target_property(original_includes existing INTERFACE_INCLUDE_DIRECTORIES)
endif()

find_package(libtcod CONFIG REQUIRED)
get_target_property(first_includes libtcod::libtcod INTERFACE_INCLUDE_DIRECTORIES)
find_package(libtcod CONFIG REQUIRED)
get_target_property(second_includes libtcod::libtcod INTERFACE_INCLUDE_DIRECTORIES)
if(NOT first_includes STREQUAL second_includes)
    message(FATAL_ERROR "Repeated find_package changed the target's include directories.")
endif()

if(TEST_EXISTING_ALIAS)
    get_target_property(alias_target libtcod::libtcod ALIASED_TARGET)
    if(NOT alias_target STREQUAL "existing" OR NOT original_includes STREQUAL second_includes)
        message(FATAL_ERROR "find_package changed the existing alias target.")
    endif()
else()
    add_executable(package_test main.c)
    target_link_libraries(package_test PRIVATE libtcod::libtcod)
endif()
